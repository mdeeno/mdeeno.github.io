{{/* FAQ Schema (FAQPage JSON-LD) Detects Q patterns in page content and generates structured data. Patterns matched: ### Q1. Question text (h3 heading) ### **Q1. Question text** (h3 heading with bold) **Q1. Question text** (bold paragraph, no heading) etc. The answer is the paragraph(s) immediately following the question line. */}} {{- if .IsPage -}} {{- $content := .RawContent -}} {{- /* Pattern 1: ### Q or ### **Q heading style */ -}} {{- $hasH3FAQ := findRE `(?m)^###\s+\*{0,2}Q\d+\.\s+.+` $content -}} {{- /* Pattern 2: **Q bold paragraph style (no heading) */ -}} {{- $hasBoldFAQ := findRE `(?m)^\*\*Q\d+\.\s+.+\*\*\s*$` $content -}} {{- $faqItems := slice -}} {{- /* ── Handle Pattern 1: ### Q heading style ── */ -}} {{- if $hasH3FAQ -}} {{- $parts := split $content "\n### Q" -}} {{- range $i, $part := $parts -}} {{- if gt $i 0 -}} {{- $lines := split $part "\n" -}} {{- $qLine := index $lines 0 -}} {{- /* Remove the "N. " prefix to get clean question text */ -}} {{- $qText := replaceRE `^\*{0,2}\d+\.\s+` "" $qLine -}} {{- $qText = replaceRE `\*{1,2}\s*$` "" $qText -}} {{- /* Collect answer lines: skip empty lines after heading, take until next heading or end */ -}} {{- $answer := "" -}} {{- $started := false -}} {{- range $j, $line := $lines -}} {{- if gt $j 0 -}} {{- $trimmed := strings.TrimSpace $line -}} {{- if and (not $started) (ne $trimmed "") -}} {{- $started = true -}} {{- end -}} {{- if $started -}} {{- /* Stop at next heading or FAQ section marker */ -}} {{- if not (hasPrefix $trimmed "#") -}} {{- if and (not (hasPrefix $trimmed "> **")) (ne $trimmed "") -}} {{- if eq $answer "" -}} {{- $answer = $trimmed -}} {{- else -}} {{- $answer = printf "%s %s" $answer $trimmed -}} {{- end -}} {{- else if hasPrefix $trimmed "#" -}} {{- break -}} {{- end -}} {{- else -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (ne $qText "") (ne $answer "") -}} {{- $item := dict "question" $qText "answer" $answer -}} {{- $faqItems = append $item $faqItems -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* ── Handle Pattern 2: **Q bold paragraph style ── */ -}} {{- if $hasBoldFAQ -}} {{- $parts := split $content "\n**Q" -}} {{- range $i, $part := $parts -}} {{- if gt $i 0 -}} {{- $lines := split $part "\n" -}} {{- $qLine := index $lines 0 -}} {{- /* Remove "N. " prefix and trailing ** */ -}} {{- $qText := replaceRE `^\d+\.\s+` "" $qLine -}} {{- $qText = replaceRE `\*{1,2}\s*$` "" $qText -}} {{- /* Collect answer lines */ -}} {{- $answer := "" -}} {{- $started := false -}} {{- range $j, $line := $lines -}} {{- if gt $j 0 -}} {{- $trimmed := strings.TrimSpace $line -}} {{- if and (not $started) (ne $trimmed "") -}} {{- $started = true -}} {{- end -}} {{- if $started -}} {{- if not (hasPrefix $trimmed "#") -}} {{- $isBoldQ := findRE `^\*\*Q\d+\.` $trimmed -}} {{- if $isBoldQ -}} {{- break -}} {{- end -}} {{- if ne $trimmed "" -}} {{- $cleanLine := replaceRE `^-\s+` "" $trimmed -}} {{- if eq $answer "" -}} {{- $answer = $cleanLine -}} {{- else -}} {{- $answer = printf "%s %s" $answer $cleanLine -}} {{- end -}} {{- end -}} {{- else -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Deduplicate: skip if this Q was already captured by Pattern 1 */ -}} {{- $isDupe := false -}} {{- range $faqItems -}} {{- if eq .question $qText -}} {{- $isDupe = true -}} {{- end -}} {{- end -}} {{- if and (ne $qText "") (ne $answer "") (not $isDupe) -}} {{- $item := dict "question" $qText "answer" $answer -}} {{- $faqItems = append $item $faqItems -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $faqItems -}} {{- end -}} {{- end -}}

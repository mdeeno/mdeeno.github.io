{{/* FAQ Schema (FAQPage JSON-LD) Source 1 (priority): front matter `faq` field faq: - q: "Question?" a: "Answer." Source 2 (fallback): Q patterns detected in page content ### Q1. Question text (h3 heading) ### **Q1. Question text** (h3 heading with bold) **Q1. Question text** (bold paragraph, no heading) */}} {{- if .IsPage -}} {{- $faqItems := slice -}} {{- /* ── Source 1: front matter faq field (highest priority) ── */ -}} {{- with .Params.faq -}} {{- range . -}} {{- if and .q .a -}} {{- $item := dict "question" .q "answer" .a -}} {{- $faqItems = append $item $faqItems -}} {{- end -}} {{- end -}} {{- end -}} {{- /* ── Source 2: content pattern detection (only if no front matter faq) ── */ -}} {{- if not $faqItems -}} {{- $content := .RawContent -}} {{- /* Pattern 1: ### Q or ### **Q heading style */ -}} {{- $hasH3FAQ := findRE `(?m)^###\s+\*{0,2}Q\d+\.\s+.+` $content -}} {{- /* Pattern 2: **Q bold paragraph style (no heading) */ -}} {{- $hasBoldFAQ := findRE `(?m)^\*\*Q\d+\.\s+.+\*\*\s*$` $content -}} {{- /* ── Handle Pattern 1: ### Q heading style ── */ -}} {{- if $hasH3FAQ -}} {{- $parts := split $content "\n### Q" -}} {{- range $i, $part := $parts -}} {{- if gt $i 0 -}} {{- $lines := split $part "\n" -}} {{- $qLine := index $lines 0 -}} {{- /* Remove the "N. " prefix to get clean question text */ -}} {{- $qText := replaceRE `^\*{0,2}\d+\.\s+` "" $qLine -}} {{- $qText = replaceRE `\*{1,2}\s*$` "" $qText -}} {{- /* Collect answer lines: skip empty lines after heading, take until next heading or end */ -}} {{- $answer := "" -}} {{- $started := false -}} {{- range $j, $line := $lines -}} {{- if gt $j 0 -}} {{- $trimmed := strings.TrimSpace $line -}} {{- if and (not $started) (ne $trimmed "") -}} {{- $started = true -}} {{- end -}} {{- if $started -}} {{- /* Stop after the answer paragraph or before non-answer blocks. */ -}} {{- if eq $trimmed "" -}} {{- break -}} {{- else if or (hasPrefix $trimmed "#") (hasPrefix $trimmed "<") (hasPrefix $trimmed "---") -}} {{- break -}} {{- else -}} {{- if not (hasPrefix $trimmed "> **") -}} {{- if eq $answer "" -}} {{- $answer = $trimmed -}} {{- else -}} {{- $answer = printf "%s %s" $answer $trimmed -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (ne $qText "") (ne $answer "") -}} {{- $item := dict "question" $qText "answer" $answer -}} {{- $faqItems = append $item $faqItems -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* ── Handle Pattern 2: **Q bold paragraph style ── */ -}} {{- if $hasBoldFAQ -}} {{- $parts := split $content "\n**Q" -}} {{- range $i, $part := $parts -}} {{- if gt $i 0 -}} {{- $lines := split $part "\n" -}} {{- $qLine := index $lines 0 -}} {{- /* Remove "N. " prefix and trailing ** */ -}} {{- $qText := replaceRE `^\d+\.\s+` "" $qLine -}} {{- $qText = replaceRE `\*{1,2}\s*$` "" $qText -}} {{- /* Collect answer lines */ -}} {{- $answer := "" -}} {{- $started := false -}} {{- range $j, $line := $lines -}} {{- if gt $j 0 -}} {{- $trimmed := strings.TrimSpace $line -}} {{- if and (not $started) (ne $trimmed "") -}} {{- $started = true -}} {{- end -}} {{- if $started -}} {{- if eq $trimmed "" -}} {{- break -}} {{- else if or (hasPrefix $trimmed "#") (hasPrefix $trimmed "<") (hasPrefix $trimmed "---") -}} {{- break -}} {{- else -}} {{- $isBoldQ := findRE `^\*\*Q\d+\.` $trimmed -}} {{- if $isBoldQ -}} {{- break -}} {{- end -}} {{- $cleanLine := replaceRE `^-\s+` "" $trimmed -}} {{- if eq $answer "" -}} {{- $answer = $cleanLine -}} {{- else -}} {{- $answer = printf "%s %s" $answer $cleanLine -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Deduplicate: skip if this Q was already captured by Pattern 1 */ -}} {{- $isDupe := false -}} {{- range $faqItems -}} {{- if eq .question $qText -}} {{- $isDupe = true -}} {{- end -}} {{- end -}} {{- if and (ne $qText "") (ne $answer "") (not $isDupe) -}} {{- $item := dict "question" $qText "answer" $answer -}} {{- $faqItems = append $item $faqItems -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}{{/* end if not $faqItems (content pattern fallback) */}} {{- if $faqItems -}} {{- end -}} {{- end -}}

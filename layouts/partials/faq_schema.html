{{/* FAQ Schema (FAQPage JSON-LD) Detects ### Q patterns in page content and generates structured data. Patterns matched: ### Q1. Question text ### Q2. Question text etc. The answer is the paragraph(s) immediately following the question heading. */}} {{- if .IsPage -}} {{- $content := .RawContent -}} {{- $hasFAQ := findRE `(?m)^###\s+Q\d+\.\s+.+` $content -}} {{- if $hasFAQ -}} {{- $questions := findRE `(?m)^###\s+Q\d+\.\s+.+` $content -}} {{- $parts := split $content "\n### Q" -}} {{- $faqItems := slice -}} {{- range $i, $part := $parts -}} {{- if gt $i 0 -}} {{- $lines := split $part "\n" -}} {{- $qLine := index $lines 0 -}} {{- /* Remove the "N. " prefix to get clean question text */ -}} {{- $qText := replaceRE `^\d+\.\s+` "" $qLine -}} {{- /* Collect answer lines: skip empty lines after heading, take until next heading or end */ -}} {{- $answer := "" -}} {{- $started := false -}} {{- range $j, $line := $lines -}} {{- if gt $j 0 -}} {{- $trimmed := strings.TrimSpace $line -}} {{- if and (not $started) (ne $trimmed "") -}} {{- $started = true -}} {{- end -}} {{- if $started -}} {{- /* Stop at next heading or FAQ section marker */ -}} {{- if not (hasPrefix $trimmed "#") -}} {{- if and (not (hasPrefix $trimmed "> **")) (ne $trimmed "") -}} {{- if eq $answer "" -}} {{- $answer = $trimmed -}} {{- else -}} {{- $answer = printf "%s %s" $answer $trimmed -}} {{- end -}} {{- else if hasPrefix $trimmed "#" -}} {{- break -}} {{- end -}} {{- else -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (ne $qText "") (ne $answer "") -}} {{- $item := dict "question" $qText "answer" $answer -}} {{- $faqItems = append $item $faqItems -}} {{- end -}} {{- end -}} {{- end -}} {{- if $faqItems -}} {{- end -}} {{- end -}} {{- end -}}
